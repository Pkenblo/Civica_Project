{{ config(unique_key=['cp', 'year']) }}

with

-- Imports
fact_crimes as (
    select
          c.*
        {% if is_incremental() %}
        , coalesce({{ after_max_ingested_at('c.ingested_at', 'crimes_ingested_at') }}, true) as is_new_ingest
        {% else %}
        , true as is_new_ingest
        {% endif %}
    from {{ ref('fact_crimes') }} c
)
, dim_date as (
    select *
    from {{ ref('dim_date') }}
)
, dim_licenses as (
    select
          l.*
        {% if is_incremental() %}
        , coalesce({{ after_max_ingested_at('l.ingested_at', 'licenses_ingested_at') }}, true) as is_new_ingest
        {% else %}
        , true as is_new_ingest
        {% endif %}
    from {{ ref('dim_licenses') }} l
)
, bridge_crime_shooting_types as (
    select *
    from {{ ref('bridge_crime_shooting_types') }}
)
, dim_address as (
    select *
    from {{ ref('dim_address') }}
)
, dim_nibrs as (
    select *
    from {{ ref('dim_nibrs') }}
)
, dim_postal_codes as (
    select *
    from {{ ref('dim_postal_codes') }}
)

-- CTEs
, crime_years as (
    select
          d.year
        , boolor_agg(c.is_new_ingest) as has_new_crimes
    from fact_crimes c
    join dim_date d
        on d.date_key = c.offense_date_key
    group by d.year
)

, license_years as (
    select
          year(license_issue_date) as year
        , boolor_agg(is_new_ingest) as has_new_licenses
    from dim_licenses
    where license_issue_date is not null
    group by year(license_issue_date)
)

, affected_years as (
    select
        c.year
    from crime_years c
    join license_years l
        on l.year = c.year
    -- Una licencia histórica cambia los acumulados desde su año de emisión.
    cross join (
        select min(year) as first_changed_year
        from license_years
        where has_new_licenses
    ) cutoff
    where c.has_new_crimes
        or c.year >= cutoff.first_changed_year
)

, shooting_crimes as (
    -- Varias etiquetas no convierten un delito en varios eventos.
    select distinct
        crime_id
    from bridge_crime_shooting_types
)

, crime_counts as (
    select
          a.cp
        , d.year
        , sum(case when n.offense_code = '240' then 1 else 0 end) as vehicle_theft_count
        , sum(case when n.offense_code in ('120', '23A', '23B', '23H') then 1 else 0 end) as theft_count
        , sum(
            case
                when s.crime_id is not null or n.offense_code in ('09A', '09B', '09C') then 1
                else 0
            end
          ) as shooting_or_homicide_count
    from fact_crimes c
    join dim_date d
        on d.date_key = c.offense_date_key
    join affected_years y
        on y.year = d.year
    join dim_address a
        on a.address_id = c.address_id
    left join dim_nibrs n
        on n.nibrs_key = c.nibrs_key
    left join shooting_crimes s
        on s.crime_id = c.crime_id
    where a.cp is not null
    group by
          a.cp
        , d.year
)

, licenses as (
    select
          cp
        , year(license_issue_date) as year
        , sum(case when species = 'DOG' then 1 else 0 end) as dog_license_count
        , sum(case when species = 'CAT' then 1 else 0 end) as cat_license_count
        , sum(case when species = 'GOAT' then 1 else 0 end) as goat_license_count
        , sum(case when species = 'PIG' then 1 else 0 end) as pig_license_count
        , count(*) as pet_license_count
    from dim_licenses
    where cp is not null
        and license_issue_date is not null
        and species in ('DOG', 'CAT', 'GOAT', 'PIG')
    group by
          cp
        , year(license_issue_date)
)

, licenses_by_cutoff as (
    -- Incluir emisiones anteriores aunque su año no tenga delitos y no sea seleccionable.
    select
          p.cp
        , y.year
        , coalesce(sum(l.dog_license_count), 0) as cumulative_dog_license_count
        , coalesce(sum(l.cat_license_count), 0) as cumulative_cat_license_count
        , coalesce(sum(l.goat_license_count), 0) as cumulative_goat_license_count
        , coalesce(sum(l.pig_license_count), 0) as cumulative_pig_license_count
    from dim_postal_codes p
    cross join affected_years y
    left join licenses l
        on l.cp = p.cp
        and l.year <= y.year
    group by
          p.cp
        , y.year
)

, counts as (
    select
          p.cp
        , y.year
        , p.geography
        , p.boundary_version
        , coalesce(c.vehicle_theft_count, 0) as vehicle_theft_count
        , coalesce(c.theft_count, 0) as theft_count
        , coalesce(c.shooting_or_homicide_count, 0) as shooting_or_homicide_count
        , coalesce(l.dog_license_count, 0) as dog_license_count
        , coalesce(l.cat_license_count, 0) as cat_license_count
        , coalesce(l.goat_license_count, 0) as goat_license_count
        , coalesce(l.pig_license_count, 0) as pig_license_count
        , coalesce(l.pet_license_count, 0) as pet_license_count
        , h.cumulative_dog_license_count
        , h.cumulative_cat_license_count
        , h.cumulative_goat_license_count
        , h.cumulative_pig_license_count
    from dim_postal_codes p
    cross join affected_years y
    left join crime_counts c
        on c.cp = p.cp
        and c.year = y.year
    left join licenses l
        on l.cp = p.cp
        and l.year = y.year
    join licenses_by_cutoff h
        on h.cp = p.cp
        and h.year = y.year
)

, source_watermarks as (
    -- Los dos orígenes se cargan de forma independiente.
    select
          (select max(ingested_at) from fact_crimes) as crimes_ingested_at
        , (select max(ingested_at) from dim_licenses) as licenses_ingested_at
)

, final as (
    select
          cp
        , year
        , vehicle_theft_count
        , theft_count
        , shooting_or_homicide_count
        , dog_license_count
        , cat_license_count
        , goat_license_count
        , pig_license_count
        , pet_license_count
        , dog_license_count / nullif(pet_license_count, 0) as dog_license_share
        , cat_license_count / nullif(pet_license_count, 0) as cat_license_share
        , goat_license_count / nullif(pet_license_count, 0) as goat_license_share
        , pig_license_count / nullif(pet_license_count, 0) as pig_license_share
        , geography
        , boundary_version
        , cumulative_dog_license_count
        , cumulative_cat_license_count
        , cumulative_goat_license_count
        , cumulative_pig_license_count
        , w.crimes_ingested_at
        , w.licenses_ingested_at
    from counts
    cross join source_watermarks w
)

select *
from final
